void PARAR(){ 
  digitalWrite(M1, LOW); digitalWrite(M1R, LOW);
  digitalWrite(M2, LOW);digitalWrite(M2R, LOW);}

void AVANZAR(){
  digitalWrite(M1, HIGH); digitalWrite(M1R, LOW);
  digitalWrite(M2, HIGH); digitalWrite(M2R, LOW);}

void RETROCEDER(){
   digitalWrite(M1, LOW); digitalWrite(M1R, HIGH);
   digitalWrite(M2, LOW); digitalWrite(M2R, HIGH);}

void DERECHA(){
  digitalWrite(M1, HIGH); digitalWrite(M1R, LOW);
  digitalWrite(M2, LOW);  digitalWrite(M2R, LOW);}

void IZQUIERDA(){
